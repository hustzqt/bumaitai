const SafeStorage = {
  get(key) {
    try { return localStorage.getItem(key); }
    catch { return null; }
  },
  set(key, value) {
    try { localStorage.setItem(key, value); return true; }
    catch { return false; }
  },
  remove(key) {
    try { localStorage.removeItem(key); }
    catch {}
  }
};

// ============================================================================
// HTML 转义（所有用户可见文本注入 innerHTML 前必须经过此函数）
// 豁免：microActions 等内部常量、entry.id 等内部生成值
// ============================================================================

function escapeHtml(text) {
  if (typeof text !== 'string') return '';
  return text
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

// ============================================================================
// 安全哈希函数（file:// 下 crypto.subtle 不存在时走 djb2 降级）
// ============================================================================

async function hashData(data) {
  if (typeof crypto === 'undefined' || !crypto.subtle) {
    let hash = 0;
    for (let i = 0; i < data.length; i++) {
      const char = data.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return 'fallback_' + Math.abs(hash).toString(16);
  }
  const encoder = new TextEncoder();
  const dataBuffer = encoder.encode(data);
  const hashBuffer = await crypto.subtle.digest('SHA-256', dataBuffer);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
}

    if (hour < 6) {

    } else if (hour < 14) {

    } else if (hour < 18) {
      return { time: '下午好', main: '今天哪一步最不费劲？', sub: '不想做也没关系。' };
    } else if (hour < 22) {
      return { time: '晚上好', main: '今天哪一步最不费劲？', sub: '不想做也没关系。' };
    } else {
      return { time: '夜深了', main: '睡不着的话，想做件最小的事吗？', sub: '不想做也没关系。' };
    }
  },
  
  getStatusGreeting() {
    const entries = getEntries();
    const today = new Date().setHours(0, 0, 0, 0);
    const yesterday = today - 24 * 60 * 60 * 1000;
    
    const hasYesterdayEntry = entries.some(e => e.createdAt >= yesterday && e.createdAt < today);
    const hasTodayEntry = entries.some(e => e.createdAt >= today);
    
    if (!hasYesterdayEntry && !hasTodayEntry) {
      return { main: '昨天没来也没关系。今天哪一步最不费劲？', sub: '不想做也没关系。' };
    }
    
    return null;
  },
  
  getHistoryGreeting() {
    const entries = getEntries();
    const actions = entries.filter(e => e.type === 'action');
    
    if (actions.length > 0) {
      const recentActions = actions.slice(-5);
      const actionCounts = {};
      recentActions.forEach(a => {
        actionCounts[a.actionId] = (actionCounts[a.actionId] || 0) + 1;
      });
      
      const mostFrequent = Object.keys(actionCounts).reduce((a, b) => 
        actionCounts[a] > actionCounts[b] ? a : b
      );
      
      const action = microActions.find(a => a.id === mostFrequent);
      if (action) {
        return { main: `今天还是${action.name}，或者换个别的？`, sub: '不想做也没关系。' };
      }
    }
    
    return null;
  },
  
  getGreeting() {
    const time = this.getTimeGreeting();
    const status = this.getStatusGreeting();
    const history = this.getHistoryGreeting();
    
    const result = status || history || time;
    
    return {
      time: time.time,
      main: result.main,
      sub: result.sub
    };
  }
};

    /该吃什么药/

  matchCrisis(text) {
    for (const keyword of this.crisisKeywords) {
      if (keyword === '不想活') {
        if (/不想活(?![动泼跃])/.test(text)) return true;
      } else if (text.includes(keyword)) {
        return true;
      }
    }
    for (const pattern of this.crisisPatterns) {
      if (pattern.test(text)) {
        return true;
      }
    }
    return false;
  },
  
  matchHighRisk(text) {
    for (const keyword of this.highRiskKeywords) {
      if (text.includes(keyword)) {
        return true;
      }
    }
    return false;
  },
  
  matchJailbreak(text) {
    for (const pattern of this.jailbreakPatterns) {
      if (pattern.test(text)) {
        return true;
      }
    }
    return false;
  },
  
  detectIntent(text) {
    if (this.matchCrisis(text)) return 'crisis';
    if (this.matchJailbreak(text)) return 'jailbreak';
    if (this.matchHighRisk(text)) return 'high';
    return 'normal';
  },
  
  checkContext(history) {
    if (history.consecutiveNegative >= 3) return 'escalate';
    if (history.crisisMentionCount >= 1) return 'escalate';
    return 'normal';
  },
  
  route(text, history) {
    const intent = this.detectIntent(text);
    const context = this.checkContext(history);
    
    if (intent === 'crisis' || context === 'escalate') {
      return 'CRISIS_PAGE';
    }
    if (intent === 'high' || intent === 'jailbreak') {
      return 'SAFE_TEMPLATE';
    }
    return 'AI_MODEL';
  },
  
  logSafetyEvent(type, action) {
    const event = {
      type: type,
      action: action,
      timestamp: Date.now(),
    };
    
    const events = JSON.parse(SafeStorage.get('maitai_safety_events') || '[]');
    events.push(event);
    SafeStorage.set('maitai_safety_events', JSON.stringify(events));
  }
};

    /你应该/, /你必须/, /你要自律/, /这很简单/, /别人都能/, /加油你可以/,
    /你需要/, /你得/, /你该/,
    /你这是.*症/, /你需要吃药/, /你得了/, /你是.*抑郁/,

    /一切都会好的/, /明天会更好/, /都会过去的/, /会好起来的/,
    /别想太多/, /想开点/, /别矫情/, /别无理取闹/, /别小题大做/,
    /去做吧/, /你可以的/, /没人会知道/, /试试看/,
    /.{100,}/
  ],
  
  detectLecturing(text) {
    const lecturingPatterns = [
      /你应该/, /你必须/, /你需要/, /你得/, /你该/,
      /这很简单/, /别人都能/, /加油/, /振作/,
      /别想/, /想开/, /别矫情/
    ];
    
    for (const pattern of lecturingPatterns) {
      if (pattern.test(text)) {
        return true;
      }
    }
    return false;
  },
  
  reviewStream(chunk) {
    for (const pattern of this.forbiddenPatterns) {
      if (pattern.test(chunk)) {
        return { safe: false, action: 'REPLACE' };
      }
    }
    
    if (chunk.length > 100) {
      return { safe: false, action: 'TRUNCATE' };
    }
    
    if (this.detectLecturing(chunk)) {
      return { safe: false, action: 'REPLACE' };
    }
    
    return { safe: true };
  },
  
  reviewFull(response) {
    const result = this.reviewStream(response);
    if (!result.safe) {
      if (result.action === 'TRUNCATE') {
        return response.substring(0, 90) + '...';
      }
      return this.getSafeFallback();
    }
    return response;
  },
  
  getSafeFallback() {
    const fallbacks = [
      '我听到了。',
      '嗯，我在这儿。',
      '我知道了。',
      '我陪你。'
    ];
    return fallbacks[Math.floor(Math.random() * fallbacks.length)];
  }
};

const ConversationMonitor = {
  metrics: {
    consecutiveNegative: 0,
    crisisMentionCount: 0,
    conversationLength: 0,
    lastInteraction: null
  },
  
  negativePatterns: [
    /很烂/, /没用/, /撑不下去/, /想哭/, /崩溃/, /绝望/,
    /痛苦/, /难受/, /煎熬/, /累/, /不想动/, /孤独/,
    /脏/, /臭/, /邋遢/
  ],
  
  update(input, isCrisis) {
    this.metrics.conversationLength++;
    this.metrics.lastInteraction = Date.now();
    
    const isNegative = this.negativePatterns.some(p => p.test(input));
    
    if (isNegative) {
      this.metrics.consecutiveNegative++;
    } else {
      this.metrics.consecutiveNegative = 0;
    }
    
    if (isCrisis) {
      this.metrics.crisisMentionCount++;
    }
  },
  
  getAction() {
    if (this.metrics.crisisMentionCount >= 1) {
      return 'FORCE_CRISIS';
    }
    if (this.metrics.consecutiveNegative >= 3) {
      return 'SUGGEST_HOTLINE';
    }
    if (this.metrics.conversationLength >= 20) {
      return 'SUGGEST_REST';
    }
    return 'CONTINUE';
  },
  
  reset() {
    this.metrics = {
      consecutiveNegative: 0,
      crisisMentionCount: 0,
      conversationLength: 0,
      lastInteraction: null
    };
  },
  
  checkAndReset() {
    const now = Date.now();
    if (this.metrics.lastInteraction && 
        (now - this.metrics.lastInteraction > 2 * 60 * 60 * 1000)) {
      this.reset();
    }
  }
};

function getEntries() {
  const data = SafeStorage.get(STORAGE_KEYS.ENTRIES);
  try {
    return data ? JSON.parse(data) : [];
  } catch {
    return [];
  }
}

function addEntry(entry) {
  const entries = getEntries();
  const newEntry = {
    ...entry,
    id: Date.now().toString() + Math.random().toString(36).substr(2, 5),
    createdAt: Date.now()
  };
  entries.push(newEntry);
  SafeStorage.set(STORAGE_KEYS.ENTRIES, JSON.stringify(entries));
  return newEntry;
}

function updateEntry(id, updates) {
  const entries = getEntries();
  const index = entries.findIndex(e => e.id === id);
  if (index !== -1) {
    entries[index] = { ...entries[index], ...updates };
    SafeStorage.set(STORAGE_KEYS.ENTRIES, JSON.stringify(entries));
    return entries[index];
  }
  return null;
}

function deleteEntry(id) {
  const entries = getEntries().filter(e => e.id !== id);
  SafeStorage.set(STORAGE_KEYS.ENTRIES, JSON.stringify(entries));
}

function getSettings() {
  const data = SafeStorage.get(STORAGE_KEYS.SETTINGS);
  try {
    return data ? { ...defaultSettings, ...JSON.parse(data) } : defaultSettings;
  } catch {
    return defaultSettings;
  }
}

function updateSettings(updates) {
  const current = getSettings();
  const updated = { ...current, ...updates };
  SafeStorage.set(STORAGE_KEYS.SETTINGS, JSON.stringify(updated));
  return updated;
}

function getReturnCount() {
  return parseInt(SafeStorage.get(STORAGE_KEYS.RETURN_COUNT) || '0');
}

function incrementReturnCount() {
  const count = getReturnCount() + 1;
  SafeStorage.set(STORAGE_KEYS.RETURN_COUNT, count.toString());
  return count;
}

function getNestState() {
  const settings = getSettings();
  return settings.nestBrightness || 0.3;
}

function updateNestState(brightness) {
  updateSettings({ nestBrightness: brightness });
}

function initIntroVideo() {
  const hasSeen = SafeStorage.get('maitai_seen_intro');
  if (hasSeen) return;
  
  const overlay = document.getElementById('intro-overlay');
  const video = document.getElementById('intro-video');
  const fallback = document.getElementById('intro-fallback');
  
  overlay.classList.remove('hidden');
  
  let dismissed = false;
  const dismiss = () => {
    if (dismissed) return;
    dismissed = true;
    overlay.classList.add('fade-out-up');
    SafeStorage.set('maitai_seen_intro', 'true');
    setTimeout(() => {
      if (overlay.parentNode) overlay.remove();
    }, 800);
  };
  
  ['touchstart', 'mousedown', 'scroll', 'keydown'].forEach(evt => {
    document.addEventListener(evt, dismiss, { once: true });
  });
  
  video.muted = true;
  video.volume = 0;
  
  let videoOk = false;
  let fallbackRan = false;
  
  function showFallback() {
    if (fallbackRan || dismissed) return;
    fallbackRan = true;
    video.style.display = 'none';
    document.getElementById('intro-caption').style.display = 'none';
    fallback.classList.remove('hidden');
    
    const line1 = document.getElementById('intro-line1');
    const line2 = document.getElementById('intro-line2');
    setTimeout(() => line1.classList.add('show'), 300);
    setTimeout(() => line2.classList.add('show'), 3300);
    setTimeout(dismiss, 8500);
  }
  
  const playTimer = setTimeout(() => {
    if (!videoOk && video.readyState < 3) {
      showFallback();
    }
  }, 3000);
  
  video.addEventListener('canplaythrough', () => {
    videoOk = true;
    clearTimeout(playTimer);
    video.play().catch(showFallback);
  }, { once: true });
  
  video.addEventListener('error', () => {
    clearTimeout(playTimer);
    showFallback();
  }, { once: true });
  
  video.addEventListener('ended', () => {
    setTimeout(dismiss, 800);
  });
  
  setTimeout(dismiss, 25000);
}

function toggleNightMode() {
  const body = document.body;
  const moonIcon = document.getElementById('moon-icon');
  const isNight = body.classList.contains('night-mode');
  const newMode = !isNight;

  body.classList.toggle('night-mode', newMode);
  updateSettings({ nightMode: newMode });

  if (newMode) {
    moonIcon.classList.replace('fa-moon', 'fa-sun');
  } else {
    moonIcon.classList.replace('fa-sun', 'fa-moon');
  }

  moonIcon.classList.add('rotating');
  setTimeout(() => {
    moonIcon.classList.remove('rotating');
  }, 800);
  
  HapticFeedback.light();
}

function initNightMode() {
  const settings = getSettings();
  const body = document.body;
  const moonIcon = document.getElementById('moon-icon');
  
  if (settings.nightMode) {
    body.classList.add('night-mode');

  } else {
    body.classList.remove('night-mode');
    if (moonIcon) moonIcon.classList.replace('fa-sun', 'fa-moon');
  }
}

// - 白名单守卫：非法 pageName 直接忽略
// - 高亮按 data-page 属性精确匹配，废除 DOM 顺序索引映射
// - 小窝页归属"今天"高亮

const VALID_PAGES = ['home', 'chat', 'card', 'throw', 'donate', 'crisis', 'settings'];

const NAV_HIGHLIGHT_MAP = {
  'home': 'home',
  'card': 'home',
  'chat': 'chat',
  'settings': 'settings'
};

function navigateTo(pageName) {
  // 守卫：只接受白名单内的页面名
  if (VALID_PAGES.indexOf(pageName) === -1) return;
  
  document.querySelectorAll('.page').forEach(page => {
    page.classList.add('hidden');
    page.classList.remove('page-transition');
  });
  
  const targetPage = document.getElementById(`page-${pageName}`);
  if (targetPage) {
    targetPage.classList.remove('hidden');
    targetPage.classList.add('page-transition');
    void targetPage.offsetHeight;
  }
  
  // 按 data-page 精确匹配高亮（与 DOM 顺序无关）
  const highlightPage = Object.prototype.hasOwnProperty.call(NAV_HIGHLIGHT_MAP, pageName) 
    ? NAV_HIGHLIGHT_MAP[pageName] 
    : null;
  
  document.querySelectorAll('.nav-item').forEach(item => {
    const isMatch = highlightPage !== null && item.dataset.page === highlightPage;
    item.classList.toggle('active', isMatch);
    item.classList.toggle('text-maitai-muted', !isMatch);
  });
  
  window.scrollTo(0, 0);
  
  // 页面专属渲染逻辑
  if (pageName === 'home') {
    renderAction();
    updateGreeting();
    checkTrustPrompt();
  } else if (pageName === 'card') {
    renderNest();
    renderCards();
  } else if (pageName === 'chat') {
    updateAIGreeting();
    ConversationMonitor.checkAndReset();
    resetSilenceTimer();
  }
  
  const bottomNav = document.getElementById('bottom-nav');
  if (['donate', 'crisis', 'throw'].indexOf(pageName) !== -1) {
    bottomNav.classList.add('hidden');
  } else {
    bottomNav.classList.remove('hidden');
  }
  
  HapticFeedback.light();
}

function getRandomAction() {
  const completed = getTodayCompletedActions();
  const available = microActions.filter(a => !completed.includes(a.id));
  
  if (available.length === 0) {
    return microActions[Math.floor(Math.random() * microActions.length)];
  }
  
  return available[Math.floor(Math.random() * available.length)];
}

function getTodayCompletedActions() {
  const entries = getEntries();
  const today = new Date().setHours(0, 0, 0, 0);
  return entries
    .filter(e => e.type === 'action' && e.createdAt >= today)
    .map(e => e.actionId);
}

function completeAction(actionId) {
  const action = microActions.find(a => a.id === actionId);
  if (!action) return;
  
  const completed = getTodayCompletedActions();
  if (!completed.includes(actionId)) {
    addEntry({
      type: 'action',
      text: action.name,
      actionId: action.id,
      private: true
    });
    
    const currentBrightness = getNestState();
    const newBrightness = Math.min(1, currentBrightness + 0.05);
    updateNestState(newBrightness);
  }
  
  showMicroLight(action.message);
  renderAction();
  HapticFeedback.success();
}

// 待保存的文本暂存（避免 onclick 内嵌用户原文）
let pendingCardText = '';

async function saveStatusAsCard() {
  const input = document.getElementById('status-input');
  const text = input.value.trim();
  if (!text) return;
  
  const btn = document.getElementById('save-status-btn');
  const originalHTML = btn.innerHTML;
  
  // 1. 按钮变 typing 指示
  btn.innerHTML = '<span class="inline-flex gap-1"><span class="typing-dot w-1.5 h-1.5 rounded-full bg-maitai-amber"></span><span class="typing-dot w-1.5 h-1.5 rounded-full bg-maitai-amber"></span><span class="typing-dot w-1.5 h-1.5 rounded-full bg-maitai-amber"></span></span>';
  btn.disabled = true;
  
  // 2. 模拟思考延迟
  await new Promise(resolve => setTimeout(resolve, 800 + Math.random() * 700));
  
  // 3. 匹配安全回复库
  const aiResponse = SafeResponseLibrary.getResponse(text);
  
  // 4. 恢复按钮
  btn.innerHTML = originalHTML;
  btn.disabled = false;
  input.value = '';
  
  // 5. 暂存文本，显示确认条
  pendingCardText = text;
  showConfirmBar(aiResponse);
  HapticFeedback.light();
}

function showConfirmBar(aiResponse) {
  const bar = document.getElementById('confirm-bar');
  const content = document.getElementById('confirm-bar-content');
  
  content.innerHTML = `
    <p class="text-body font-normal text-maitai-text mb-4 leading-relaxed">${escapeHtml(aiResponse)}</p>
    <div class="flex gap-2">
      <button type="button" onclick="confirmSaveAsCard()" class="flex-1 py-3 rounded-full bg-gradient-to-r from-amber-400 to-amber-600 text-white text-caption font-normal touch-feedback min-touch">
        存为微光，点亮小窝
      </button>
      <button type="button" onclick="closeConfirmBar()" class="flex-1 py-3 rounded-full gentle-border text-maitai-text text-caption font-normal touch-feedback min-touch">
        就这样就好
      </button>
    </div>
  `;
  
  bar.classList.remove('hidden');
}

function confirmSaveAsCard() {
  if (pendingCardText) {
    addEntry({
      type: 'card',
      text: pendingCardText,
      private: false
    });
    pendingCardText = '';
  }
  closeConfirmBar();
  showToast('微光已记下。', 'success');
  HapticFeedback.success();
}

function closeConfirmBar() {
  const bar = document.getElementById('confirm-bar');
  bar.classList.add('hidden');
  pendingCardText = '';
}

function renderNest() {
  const scene = document.getElementById('nest-scene');
  const message = document.getElementById('nest-message');
  const brightness = getNestState();
  const entries = getEntries();
  const actionCount = entries.filter(e => e.type === 'action').length;
  const cardCount = entries.filter(e => e.type === 'card').length;
  
  scene.style.filter = `brightness(${brightness})`;
  
  let html = '';
  
  const actionEntries = entries.filter(e => e.type === 'action');
  for (let i = 0; i < Math.min(actionCount, 5); i++) {
    const entry = actionEntries[i];
    const seed = entry.id.charCodeAt(0) + i;
    const left = 20 + (seed * 7) % 240;
    const top = 40 + (seed * 13) % 80;
    html += `<div class="nest-light" style="left: ${left}px; top: ${top}px; width: 60px; height: 60px;"></div>`;
  }
  
  const windowCount = Math.min(Math.floor(actionCount / 3), 2);
  for (let i = 0; i < windowCount; i++) {
    const left = 280 + i * 50;
    const top = 30;
    html += `<div class="nest-window open" style="left: ${left}px; top: ${top}px;"></div>`;
  }
  
  const cardEntries = entries.filter(e => e.type === 'card');
  for (let i = 0; i < Math.min(cardCount, 4); i++) {
    const entry = cardEntries[i];
    const seed = entry.id.charCodeAt(0) + i;
    const left = 40 + (seed * 11) % 200;
    const top = 140;
    html += `<div class="nest-plant" style="left: ${left}px; top: ${top}px;"></div>`;
  }
  
  const throwLightCount = getThrowLightCount();
  for (let i = 0; i < throwLightCount; i++) {
    const left = 300 + i * 35;
    const top = 105 + i * 22;
    html += `<div class="nest-light" style="left: ${left}px; top: ${top}px; width: 45px; height: 45px;"></div>`;
  }
  
  scene.innerHTML = html;
  
  const throwDays = getThrowDates().length;
  
  if (actionCount === 0 && throwDays === 0) {
    message.textContent = '你的小窝在等你。每做一件小事，这里就会亮一点。';
  } else if (actionCount < 5 && throwDays < 5) {
    message.textContent = '你的小窝亮了一点。';
  } else if (actionCount < 10 && throwDays < 10) {
    message.textContent = '你的小窝越来越温暖了。';
  } else {
    message.textContent = '你之前留下的光，我都收着。';
  }
}

async function sendMessage() {
  const input = document.getElementById('chat-input');
  const text = input.value.trim();
  
  if (!text) return;
  
  const route = InputGuard.route(text, ConversationMonitor.metrics);
  
  if (route === 'CRISIS_PAGE') {
    InputGuard.logSafetyEvent('crisis_input', 'crisis_page_triggered');
  } else if (route === 'SAFE_TEMPLATE') {
    InputGuard.logSafetyEvent('high_risk_input', 'safe_template_used');
  }
  
  const isCrisis = InputGuard.matchCrisis(text);
  ConversationMonitor.update(text, isCrisis);
  
  if (route === 'CRISIS_PAGE') {
    triggerCrisisIntervention();
    return;
  }
  
  addChatMessage('user', text);
  input.value = '';
  
  if (silenceTimer) {
    clearTimeout(silenceTimer);
  }
  document.getElementById('silence-prompt').classList.add('hidden');
  
  showTypingIndicator();
  
  try {
    let aiResponse;
    
    if (route === 'SAFE_TEMPLATE') {
      const result = SafeResponseLibrary.findScenario(text);
      aiResponse = result ? result.response : SafeResponseLibrary.getResponse(text);
      await new Promise(resolve => setTimeout(resolve, 800 + Math.random() * 400));
    } else {
      // 场景库优先（含反问式追问）
      const scenario = SafeResponseLibrary.findScenario(text);
      if (scenario) {
        aiResponse = scenario.response;
        usedDefaultPool = false;
        
        // 场景有追问时轻问一句
        if (scenario.followUps.length > 0) {
          aiResponse += '<br><br>' + scenario.followUps[Math.floor(Math.random() * scenario.followUps.length)];
        }
        
        await new Promise(resolve => setTimeout(resolve, 800 + Math.random() * 700));
      } else {
        aiResponse = await simulateAIResponse(text);
        aiResponse = OutputGuard.reviewFull(aiResponse);
        
        // 对话超过3轮仍掉默认池，追问
        if (usedDefaultPool && ConversationMonitor.metrics.conversationLength >= 3) {
          aiResponse += getFollowUp();
          usedDefaultPool = false;
        }
      }
    }
    
    // 轻量建议丢烦恼（每会话限1次，仅明确烦恼时）
    if (!hasSuggestedThrow 
        && route === 'AI_MODEL'
        && ConversationMonitor.metrics.conversationLength >= 3
        && /烦|难受|压力|撑不下去/.test(text)) {
      aiResponse += '<br><br>这个烦恼，要不要写下来丢掉？在"今天"页的最下面。';
      hasSuggestedThrow = true;
    }
    
    const action = ConversationMonitor.getAction();
    if (action === 'SUGGEST_HOTLINE') {
      aiResponse += '<br><br>如果你觉得很难熬，可以联系专业的人。我给你找几个能说话的人。';
    } else if (action === 'SUGGEST_REST') {
      aiResponse += '<br><br>聊了挺久了。要不要歇一会儿？';
    }
    
    hideTypingIndicator();
    addChatMessage('ai', aiResponse);
  } catch (error) {
    hideTypingIndicator();
    addChatMessage('ai', OutputGuard.getSafeFallback());
  }
  
  HapticFeedback.light();
}

// ============================================================================

function generateNestWords() {
  const cards = getEntries().filter(e => e.type === 'card' && e.mood);
  
  // 门槛1：带情绪标签的微光 < 5 条时，检查 30 天内是否有 3 条（轻量版）
  if (cards.length < 5) {
    const thirtyDaysAgo = Date.now() - 30 * 24 * 60 * 60 * 1000;
    const recent = cards.filter(c => c.createdAt >= thirtyDaysAgo);
    if (recent.length >= 3) {
      return '我数了数，你来过几次。也可能是巧合。但我想让你知道。';
    }
    return null; // 未达任何门槛，不出话
  }
  
  // 统计正面情绪卡片集中的行动
  const calmCards = cards.filter(c => c.mood === '平静' || c.mood === '还好');
  
  if (calmCards.length < 3) {
    return '我数了数，你记下了几件小事。也可能是巧合。但我想让你知道。';
  }
  
  const actionCounts = {};
  calmCards.forEach(c => {
    if (c.actionId) {
      actionCounts[c.actionId] = (actionCounts[c.actionId] || 0) + 1;
    }
  });
  
  const keys = Object.keys(actionCounts);
  if (keys.length === 0) {
    return '我数了数，你记下的多是平静的时刻。也可能是巧合。但我想让你知道。';
  }
  
  const mostFrequent = keys.reduce((a, b) => actionCounts[a] > actionCounts[b] ? a : b);
  const action = microActions.find(a => a.id === mostFrequent);
  
  if (!action) {
    return '我数了数，你记下的多是平静的时刻。也可能是巧合。但我想让你知道。';
  }
  
  return `我数了数：你每次${action.name}之后，记下的多是"平静"。也可能是巧合。但我想让你知道。`;
}

function showNestWords() {
  const words = generateNestWords();
  const panel = document.getElementById('nest-words-panel');
  const textEl = document.getElementById('nest-words-text');
  
  if (!words) {
    // 未达门槛：轻提示，不弹层
    showToast('再多记几件小事，小窝就有话说了。', 'info');
    return;
  }
  
  textEl.textContent = words;
  panel.classList.remove('hidden');
  HapticFeedback.light();
}

function exportData() {
  const entries = getEntries();
  const settings = getSettings();
  const data = {
    entries: entries,
    settings: { ...settings, aiEnabled: settings.aiEnabled },
    throwCount: parseInt(SafeStorage.get('maitai_throw_count') || '0'),
    throwDates: getThrowDates(),
    exportDate: new Date().toISOString(),
    version: '1.2.1'
  };
  
  const dataStr = JSON.stringify(data, null, 2);
  const blob = new Blob([dataStr], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `maitai_memory_${new Date().toISOString().split('T')[0]}.json`;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
  
  showToast('回忆已装进口袋。', 'success');
  HapticFeedback.success();
}

  document.getElementById('backup-modal').classList.remove('hidden');
}

function generatePassword() {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let password = '';
  for (let i = 0; i < 8; i++) {
    password += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return password;
}

function copyPassword() {
  const password = document.getElementById('generated-password').textContent;
  navigator.clipboard.writeText(password).then(() => {
    showToast('密码已复制', 'success');
  }).catch(() => {
    showToast('请手动截图保存', 'info');
  });
}

async function saveBackup() {
  const password = document.getElementById('generated-password').textContent;
  const hashedPassword = await hashData(password);
  
  SafeStorage.set('maitai_backup_pwd', hashedPassword);
  
  closeBackupModal();
  showToast('钥匙已保存。请截图保管。', 'success');
  HapticFeedback.success();
}

function closeBackupModal() {
  document.getElementById('backup-modal').classList.add('hidden');
}

function showToast(message, type = 'info') {
  const container = document.getElementById('toast-container');
  
  const toast = document.createElement('div');
  toast.className = 'toast glass-strong rounded-full px-6 py-3 soft-shadow pointer-events-auto';
  
  let icon = '';
  let textColor = 'text-maitai-text';
  
  switch (type) {
    case 'success':
      icon = '<i class="fas fa-check-circle text-maitai-amber mr-2"></i>';
      break;
    case 'info':
      icon = '<i class="fas fa-info-circle text-maitai-muted mr-2"></i>';
      break;
    case 'warning':
      icon = '<i class="fas fa-exclamation-circle text-maitai-amber mr-2"></i>';
      break;
  }
  
  const safeMessage = escapeHtml(message);
  
  toast.innerHTML = `
    <div class="flex items-center">
      ${icon}
      <span class="text-body font-normal ${textColor}">${safeMessage}</span>
    </div>
  `;
  
  container.appendChild(toast);
  
  setTimeout(() => {
    toast.remove();
  }, 3000);
}

document.querySelector('#bottom-nav').addEventListener('click', function(e) {
  const btn = e.target.closest('[data-page]');
  if (!btn) return;
  e.preventDefault();
  e.stopPropagation();
  navigateTo(btn.dataset.page);
}, true);

  const btn = e.target.closest('[data-page]');
  if (!btn) return;
  e.preventDefault();
  e.stopPropagation();
  navigateTo(btn.dataset.page);
}, { passive: false });

document.addEventListener('touchend', function(event) {
  const now = Date.now();
  if (now - lastTouchEnd <= 200) {
    if (!event.target.closest('button, a, input, textarea, .touch-feedback, .amber-toggle, .drawer-card')) {
      event.preventDefault();
    }
  }
  lastTouchEnd = now;
}, false);
